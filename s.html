Krzywa Sierpińskiego

Patrząc na krzywe poziomów drugiego i wyżej można dojść do wniosku że krzywa poziomu pierwszego jest podstawowym budulcem wyższych poziomów. Nie jest to jednak prawda. Najważniejszym spostrzeżeniem jest to, że — w przeciwieństwie do krzywych Hilberta — krzywe Sierpińskiego są zamknięte (można by je zamalować w środku bez obawy o zamazanie całego obrazu). Jak się okazuje, bazowy plan rekurencji sprowadza się do kreślenia krzywych otwartych, natomiast powiązanie czterech części składowych za pomocą łącz nie należy już do schematu rekurencji (podwójna strzałka oznacza podwójną długość):

A: A↘B⟹D↗A
B: B↙C⇓A↘B
C: C↖D⟸B↙C
D: D↗A⇑C↖D

Warunki początkowe:
S: A↘B↙C↖D↗

Dane:

  • n — numer krzywej
  • h — długość odcinka elementarnego
  • x, y — współrzędne punktu początkowego
Narzędzia:
  • Canvas.moveTo(x, y) — ustalenie punktu początkowego (x, y)
  • Canvas.lineTo(u, v) — kreślenie odcinka od punktu (x, y) do punktu (u, v)
Realizacja krzywych Sierpińskiego w języku JavaScript. W porównaniu z powyższą prezentacją kod uproszczono o centrowanie, ustalanie odstępów i wielokolorowość:
var c = document.getElementById("canvas"),
    n = 2,
    h = 10,
    x = 0,
    y = 0;

A(n); c.moveTo(x, y); x += h; y -= h; c.lineTo(x, y);
B(n); c.moveTo(x, y); x -= h; y -= h; c.lineTo(x, y);
C(n); c.moveTo(x, y); x -= h; y += h; c.lineTo(x, y);
D(n); c.moveTo(x, y); x += h; y += h; c.lineTo(x, y); 

function A(n) {
	if (n == 0) return;
	A(n - 1); c.moveTo(x, y); x += h; y -= h; c.lineTo(x, y);
	B(n - 1); c.moveTo(x, y); x += 2 * h; c.lineTo(x, y);
	D(n - 1); c.moveTo(x, y); x += h; y += h; c.lineTo(x, y);
	A(n - 1);
}
function B(n) {
	if (n == 0) return;
	B(n - 1); c.moveTo(x, y); x -= h; y -= h; c.lineTo(x, y);
	C(n - 1); c.moveTo(x, y); y -= 2 * h; c.lineTo(x, y);
	A(n - 1); c.moveTo(x, y); x += h; y -= h; c.lineTo(x, y);
	B(n - 1);
}
function C(n) {
	if (n == 0) return;
	C(n - 1); c.moveTo(x, y); x -= h; y += h; c.lineTo(x, y); 
	D(n - 1); c.moveTo(x, y); x -= 2 * h; c.lineTo(x, y);
	B(n - 1); c.moveTo(x, y); x -= h; y -= h; c.lineTo(x, y);
	C(n - 1);
}
function D(n) {
	if (n == 0) return;
	D(n - 1); c.moveTo(x, y); x += h; y += h; c.lineTo(x, y);
	A(n - 1); c.moveTo(x, y); y += 2 * h; c.lineTo(x, y);
	C(n - 1); c.moveTo(x, y); x -= h; y += h; c.lineTo(x, y);
	D(n - 1);
}

c.stroke();
